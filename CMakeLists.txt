cmake_minimum_required(VERSION 3.24)
project(NekoLauncher VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent)
find_package(LibArchive REQUIRED)
find_package(Qt6Keychain REQUIRED)
find_package(Qt6QWebpPlugin REQUIRED PATH_SUFFIXES lib/cmake/Qt6Gui)
add_library(neko_core
 src/core.cpp src/core.hpp src/instance_services.cpp src/instance_services.hpp src/mod_storage.cpp src/mod_storage.hpp src/file_ops.cpp src/file_ops.hpp
 src/core/archive.cpp src/core/http.cpp src/core/metadata.cpp
 src/core/loaders.cpp src/core/runtime.cpp src/core/launch.cpp
 src/core/packs.cpp src/core/mods.cpp src/core/auth.cpp
)
target_link_libraries(neko_core PUBLIC Qt6::Network LibArchive::LibArchive)
target_include_directories(neko_core PUBLIC src)
if(UNIX AND NOT APPLE)
 find_library(NEKO_VULKAN_LIBRARY NAMES vulkan)
 set(NEKO_NATIVE_LIBRARY_DIRS "")
 foreach(NEKO_NATIVE_NAME IN ITEMS wayland-client xkbcommon EGL GL X11 Xcursor Xrandr Xi Xfixes pulse pipewire-0.3 asound)
  string(MAKE_C_IDENTIFIER "${NEKO_NATIVE_NAME}" NEKO_NATIVE_ID)
  find_library(NEKO_${NEKO_NATIVE_ID}_LIBRARY NAMES ${NEKO_NATIVE_NAME})
  if(NEKO_${NEKO_NATIVE_ID}_LIBRARY)
   get_filename_component(NEKO_NATIVE_DIR "${NEKO_${NEKO_NATIVE_ID}_LIBRARY}" DIRECTORY)
   list(APPEND NEKO_NATIVE_LIBRARY_DIRS "${NEKO_NATIVE_DIR}")
  endif()
 endforeach()
 list(REMOVE_DUPLICATES NEKO_NATIVE_LIBRARY_DIRS)
 list(JOIN NEKO_NATIVE_LIBRARY_DIRS ":" NEKO_NATIVE_PATHS)
 target_compile_definitions(neko_core PRIVATE NEKO_NATIVE_LIBRARY_PATHS="${NEKO_NATIVE_PATHS}")
 if(NEKO_VULKAN_LIBRARY)
  get_filename_component(NEKO_VULKAN_LIBRARY_DIR "${NEKO_VULKAN_LIBRARY}" DIRECTORY)
  target_compile_definitions(neko_core PRIVATE NEKO_VULKAN_LIBRARY_PATH="${NEKO_VULKAN_LIBRARY_DIR}")
 endif()
endif()
add_executable(neko-launcher
 src/main.cpp src/window.cpp src/window.hpp src/design.cpp src/design.hpp src/cards.cpp src/cards.hpp
 src/ui/widgets.hpp src/ui/library.cpp src/ui/discover.cpp src/ui/accounts.cpp
 src/ui/settings.cpp src/ui/appearance.cpp src/ui/instance_editor.cpp src/ui/mods.cpp
 src/ui/storage.cpp src/ui/storage.hpp src/ui/mod_browser.cpp src/ui/mod_browser.hpp src/ui/import.cpp src/ui/launch.cpp src/ui/jobs.cpp src/ui/verification.cpp
 src/ui/instance_tools.cpp src/ui/instance_tools.hpp src/ui/command_palette.cpp src/ui/command_palette.hpp
)
target_compile_definitions(neko-launcher PRIVATE NEKO_IMAGE_PLUGIN_PATH="$<TARGET_FILE_DIR:Qt6::QWebpPlugin>/..")
target_link_libraries(neko-launcher PRIVATE neko_core Qt6::Widgets Qt6::Concurrent qt6keychain)
if(MSVC)
 target_compile_options(neko_core PRIVATE /W4)
else()
 target_compile_options(neko_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(core-tests tests/core_tests.cpp)
 target_link_libraries(core-tests PRIVATE neko_core)
 add_test(NAME core-tests COMMAND core-tests)
 add_executable(instance-services-tests tests/instance_services_tests.cpp)
 target_link_libraries(instance-services-tests PRIVATE neko_core)
 add_test(NAME instance-services-tests COMMAND instance-services-tests)
 add_executable(mod-install-tests tests/mod_install_tests.cpp)
 target_link_libraries(mod-install-tests PRIVATE neko_core)
 add_test(NAME mod-install-tests COMMAND mod-install-tests)
 add_executable(mod-storage-tests tests/mod_storage_tests.cpp)
 target_link_libraries(mod-storage-tests PRIVATE neko_core)
 add_test(NAME mod-storage-tests COMMAND mod-storage-tests)
 add_executable(audio-tests tests/audio_tests.cpp)
 target_link_libraries(audio-tests PRIVATE neko_core)
 add_executable(keychain-tests tests/keychain_tests.cpp)
 target_link_libraries(keychain-tests PRIVATE Qt6::Core qt6keychain)
endif()
install(TARGETS neko-launcher RUNTIME DESTINATION bin)
